កូដ Google Apps Script (Code.gs) សម្រាប់ឯកសារ test-1 / Sheet1 
const SPREADSHEET_ID = '10JSUA_er-tnyzaTiSTyPq0b8Ovkwt77mT9GJ1HGVKDo';
const SHEET_NAME = 'Sheet1';
const HEADERS = ['ល.រ', 'ឈ្មោះអ្នកចូលរួម', 'ទឹកប្រាក់ ($)', 'ទឹកប្រាក់ (៛)', 'ប្រតិបត្តិការ', 'ម៉ោងកត់ត្រា', 'ID'];

function getTargetSheet_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
    const headerRange = sheet.getRange(1, 1, 1, HEADERS.length);
    headerRange.setFontWeight('bold');
    headerRange.setBackground('#C7F249');
    headerRange.setFontColor('#1E1F1D');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function jsonOut_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function renumberSerials_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;
  const serials = [];
  for (let i = 1; i <= lastRow - 1; i++) {
    serials.push([i]);
  }
  sheet.getRange(2, 1, serials.length, 1).setValues(serials);
}

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const sheet = getTargetSheet_();
    const params = (e && e.parameter) ? e.parameter : {};
    let body = {};
    if (e && e.postData && e.postData.contents) {
      try {
        body = JSON.parse(e.postData.contents);
      } catch (parseErr) {
        body = {};
      }
    }

    const action = String(body.action || params.action || 'read').toLowerCase();
    const rec = body.record || body;

    if (action === 'init') {
      return jsonOut_({
        status: 'ok',
        action: 'init',
        spreadsheetId: SPREADSHEET_ID,
        sheetName: sheet.getName(),
        headers: HEADERS
      });
    }

    if (action === 'read') {
      const values = sheet.getDataRange().getValues();
      if (values.length <= 1) {
        return jsonOut_({
          status: 'ok',
          spreadsheetId: SPREADSHEET_ID,
          sheetName: sheet.getName(),
          records: []
        });
      }
      const headers = values[0].map(String);
      const records = [];
      for (let i = 1; i < values.length; i++) {
        const row = values[i];
        if (!row[1] && !row[6]) continue;
        const item = {};
        headers.forEach((h, idx) => { item[h] = row[idx]; });
        records.push(item);
      }
      return jsonOut_({
        status: 'ok',
        spreadsheetId: SPREADSHEET_ID,
        sheetName: sheet.getName(),
        records: records
      });
    }

    if (action === 'sync_all' && Array.isArray(body.records)) {
      sheet.clearContents();
      sheet.appendRow(HEADERS);
      const headerRange = sheet.getRange(1, 1, 1, HEADERS.length);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#C7F249');
      headerRange.setFontColor('#1E1F1D');
      sheet.setFrozenRows(1);

      const rows = body.records.map((r, idx) => {
        const curr = String(r.currency || 'USD').toUpperCase();
        const amt = Number(r.amount || 0);
        const usd = r.usd !== undefined ? Number(r.usd) : (curr === 'USD' ? amt : 0);
        const khr = r.khr !== undefined ? Number(r.khr) : (curr === 'KHR' ? amt : 0);
        return [
          idx + 1,
          r['ឈ្មោះអ្នកចូលរួម'] || r.name || '',
          usd,
          khr,
          r['ប្រតិបត្តិការ'] || r.method || 'សាច់ប្រាក់',
          r['ម៉ោងកត់ត្រា'] || r.timestamp || r.date || '',
          r['ID'] || r.id || Utilities.getUuid()
        ];
      });
      if (rows.length > 0) {
        sheet.getRange(2, 1, rows.length, HEADERS.length).setValues(rows);
      }
      return jsonOut_({ status: 'ok', action: 'sync_all', count: rows.length });
    }

    if (action === 'add') {
      const id = String(rec['ID'] || rec.id || params.id || Utilities.getUuid());
      const name = String(rec['ឈ្មោះអ្នកចូលរួម'] || rec.name || params.name || '').trim();
      const curr = String(rec.currency || params.currency || 'USD').toUpperCase();
      const amt = Number(rec.amount || params.amount || 0);
      const usd = (rec['ទឹកប្រាក់ ($)'] !== undefined || rec.usd !== undefined || params.usd !== undefined)
        ? Number(rec['ទឹកប្រាក់ ($)'] ?? rec.usd ?? params.usd ?? 0)
        : (curr === 'USD' ? amt : 0);
      const khr = (rec['ទឹកប្រាក់ (៛)'] !== undefined || rec.khr !== undefined || params.khr !== undefined)
        ? Number(rec['ទឹកប្រាក់ (៛)'] ?? rec.khr ?? params.khr ?? 0)
        : (curr === 'KHR' ? amt : 0);
      const method = String(rec['ប្រតិបត្តិការ'] || rec.method || params.method || 'សាច់ប្រាក់');
      const ts = String(rec['ម៉ោងកត់ត្រា'] || rec.timestamp || rec.date || params.timestamp || '');

      const existing = sheet.getDataRange().getValues();
      for (let i = 1; i < existing.length; i++) {
        if (String(existing[i][6]) === id) {
          return jsonOut_({ status: 'ok', action: 'add', duplicate: true, id: id });
        }
      }

      const nextSerial = Math.max(1, sheet.getLastRow());
      sheet.appendRow([nextSerial, name, usd, khr, method, ts, id]);
      return jsonOut_({ status: 'ok', action: 'add', id: id, rowNumber: nextSerial });
    }

    if (action === 'update') {
      const targetId = String(rec['ID'] || rec.id || params.id || '');
      const name = String(rec['ឈ្មោះអ្នកចូលរួម'] || rec.name || params.name || '').trim();
      const curr = String(rec.currency || params.currency || 'USD').toUpperCase();
      const amt = Number(rec.amount || params.amount || 0);
      const usd = (rec['ទឹកប្រាក់ ($)'] !== undefined || rec.usd !== undefined || params.usd !== undefined)
        ? Number(rec['ទឹកប្រាក់ ($)'] ?? rec.usd ?? params.usd ?? 0)
        : (curr === 'USD' ? amt : 0);
      const khr = (rec['ទឹកប្រាក់ (៛)'] !== undefined || rec.khr !== undefined || params.khr !== undefined)
        ? Number(rec['ទឹកប្រាក់ (៛)'] ?? rec.khr ?? params.khr ?? 0)
        : (curr === 'KHR' ? amt : 0);
      const method = String(rec['ប្រតិបត្តិការ'] || rec.method || params.method || 'សាច់ប្រាក់');
      const ts = String(rec['ម៉ោងកត់ត្រា'] || rec.timestamp || rec.date || params.timestamp || '');

      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][6]) === targetId) {
          sheet.getRange(i + 1, 2, 1, 6).setValues([[
            name,
            usd,
            khr,
            method,
            ts || data[i][5],
            targetId
          ]]);
          break;
        }
      }
      return jsonOut_({ status: 'ok', action: 'update', id: targetId });
    }

    if (action === 'delete') {
      const targetId = String(body.id || rec['ID'] || rec.id || params.id || '');
      const data = sheet.getDataRange().getValues();
      for (let i = data.length - 1; i >= 1; i--) {
        if (String(data[i][6]) === targetId) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      renumberSerials_(sheet);
      return jsonOut_({ status: 'ok', action: 'delete', id: targetId });
    }

    return jsonOut_({ status: 'ok', action: action });
  } catch (err) {
    return jsonOut_({ status: 'error', message: String(err) });
  } finally {
    lock.releaseLock();
  }
}
